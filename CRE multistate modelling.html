duplicate_NA_rows <- function(x, colNA = "estimate",
                              col_sel = c("denominator", "numerator", "estimate",
                                          "lower", "upper")) {
  x <- mutate(x, .id = row_number())
  NA_ind <- firstNA(x[[colNA]])
  duplicates <- x[NA_ind, ]

get_nN <- function(var, data) {
  
  x <- data[[var]]
  
  # binary (logical or 0/1)
  if (is.logical(x) || (is.numeric(x) && all(x %in% c(0,1,NA)))) {
    n <- sum(x == TRUE, na.rm = TRUE)
    N <- sum(!is.na(x))
    
    return(data.frame(
      variable = paste0(var, "TRUE"),
      n = n,
      N = N,
      pct = n / N * 100
    ))
  }
  
  # factor / character
  if (is.factor(x) || is.character(x)) {
    tab <- table(x, useNA = "no")
    N <- sum(tab)
    
    return(data.frame(
      variable = paste0(var, names(tab)),
      n = as.numeric(tab),
      N = N,
      pct = as.numeric(tab) / N * 100
    ))
  }
  
  # continuous → return NULL (handled separately)
  return(NULL)
}

A function to extract HR from msm to a table format

extract_hr_msm <- function(fit) {
  
  s <- summary(fit)
  
  if (is.null(s$hazard)) return(NULL)
  
  hazard_list <- s$hazard
  
  results_list <- list()
  
  for (name in names(hazard_list)) {
    
    mat <- as.data.frame(hazard_list[[name]])
    
    # keep only 1→2 transition
    mat <- mat[grep("State 1 - State 2", rownames(mat)), , drop = FALSE]
    
    # skip if empty or missing columns
    if (nrow(mat) == 0 || !all(c("HR", "L", "U") %in% colnames(mat))) next
    
    results_list[[name]] <- data.frame(
      HR = mat$HR[1],
      lower = mat$L[1],
      upper = mat$U[1]
    )
  }
  
  if (length(results_list) == 0) return(NULL)
  
  bind_rows(results_list)
}

5.3 Enrolled

A function that computes the data for ward occupancy:

ward_occupancy <- function(x) {
  f <- function(...) pivot_longer(..., names_to = "change", values_to = "date")
  bind_rows(f(select(x, -DISCHARGE), ADMISSION),
            f(select(x, -ADMISSION), DISCHARGE)) |> 
    arrange(date) |> 
    mutate(across(change, ~ setNames(c(1, -1), c("ADMISSION", "DISCHARGE"))[.x])) |> 
    group_by(date) |> 
    summarise(change = sum(change)) |> 
    ungroup() |> 
    mutate(occupancy = cumsum(change)) |> 
    select(-change)
}

A function that plots a ward occupancy:

plot_occ <- function(x, y = NULL, ...) {
  x |> 
    first() |> 
    mutate(occupancy = 0) |> 
    bind_rows(x) |> 
    with({
      plot(date, occupancy, type = "n", ...)
      polygon(c(head(date, 2), rep(tail(date, -2), each = 2)),
              c(first(occupancy),
                rep(tail(head(occupancy, -1), -1), each = 2),
                last(occupancy)), col = "grey")
      })
  if (! is.null(y)) abline(v = y, col = "red")
}

safe_binom_test <- function(x, n) {
  if (is.na(x) || is.na(n) || n <= 0 || x < 0 || x > n) {
    return(tibble(
      estimate = NA_real_,
      conf.low = NA_real_,
      conf.high = NA_real_,
      p.value = NA_real_
    ))
  }

  bt <- binom.test(x, n)

  tibble(
    estimate = unname(bt$estimate),
    conf.low = bt$conf.int[1],
    conf.high = bt$conf.int[2],
    p.value = bt$p.value
  )
}

presence_abscence <- function(x, id, time_point, identification, bacteria) {

}

    mutate("{{discharge}}" := as.numeric({{ discharge }} - {{ admission }}) / mpd,
           "{{admission}}" := 0) |> 
    pivot_longer(- {{ id }},
                 names_to = as_name(enquo(time_point)), values_to = "days") |>
    left_join(x = {{ state }}, y = _,
              c(as_name(enquo(id)), as_name(enquo(time_point))))
}

msm_data <- function(x, y, id, time_point, identification, bacteria,
                     admission, discharge) {
  x |>

add_CRE_prevalence_in_ward <- function(x) {
  ward_means <- function(y) {
    total_duration <- time_diff(range(y$date))
    y |> 
      mutate(weight = time_diff(c(date, NA)) / total_duration) |> 
      summarise(across(c(estimate, lower, upper),

9.2 Multi-state modelling

9.2.1 State definition and transition structure

We defined a two-state continuous-time multistate model with the following states:

State 1: No carriage of CRE
State 2: Carriage of CRE

The following transitions were allowed:

1 → 2: Acquisition of CRE carriage
2 → 1: Clearance of CRE carriage

Patient states were observed at discrete time points (admission and discharge), and transitions between states were not observed continuously, resulting in interval-censored data. The underlying process was assumed to follow a time-homogeneous continuous-time Markov model.

Clearance (2 → 1) was included to allow reversibility of carriage status between observation times, although the exact timing of clearance was not observed. Given the limited number of observation time points, estimates of clearance should be interpreted with caution.

Time was measured in days since admission. Discharge defined the end of the observation period and was treated as right censoring rather than as a separate state. Transition intensities are therefore interpreted conditional on the observed follow-up time. Because discharge terminates observation and may be related to patient status, it may represent an informative censoring mechanism, which is not explicitly modeled in this framework.

10.2 Multi-state modelling

10.2.1 State definition and transition structure

We defined a two-state continuous-time multistate model with the following states:

State 1: No carriage of CRE
State 2: Carriage of CRE

The following transitions were allowed:

1 → 2: Acquisition of CRE carriage
2 → 1: Clearance of CRE carriage

Patient states were observed at discrete time points (admission and discharge), and transitions between states were not observed continuously, resulting in interval-censored data. The underlying process was assumed to follow a time-homogeneous continuous-time Markov model.

Clearance (2 → 1) was included to allow reversibility of carriage status between observation times, although the exact timing of clearance was not observed. Given the limited number of observation time points, estimates of clearance should be interpreted with caution.

Time was measured in days since admission. Discharge defined the end of the observation period and was treated as right censoring rather than as a separate state. Transition intensities are therefore interpreted conditional on the observed follow-up time. Because discharge terminates observation and may be related to patient status, it may represent an informative censoring mechanism, which is not explicitly modeled in this framework.
